map of study locations

study_locations <- all_fields %>%
  filter(!is.na(lat), !is.na(lon)) %>%
  group_by(field) %>%
  summarise(lat = mean(lat), lon = mean(lon), .groups = "drop")

germany <- ne_countries(scale = "medium", country = "Germany", returnclass = "sf")

ggplot() +
  geom_sf(data = germany, fill = "grey95") +
  geom_point(data = study_locations, aes(x = lon, y = lat, color = field), size = 3) +
  theme_minimal() +
  labs(title = "", x = "Longitude", y = "Latitude", color = "Field")

na overview

na_by_field <- all_fields %>%
  group_by(field) %>%
  summarise(across(everything(), ~ mean(is.na(.)))) %>%
  pivot_longer(-field, names_to = "column", values_to = "na_share")

ggplot(na_by_field, aes(x = column, y = field, fill = na_share)) +
  geom_tile() +
  scale_fill_gradient(low = "white", high = "red") +
  theme_minimal() +
  theme(axis.text.x = element_text(angle = 90, hjust = 1)) +
  labs(title = "share of na per column, by study", fill = "na share")
